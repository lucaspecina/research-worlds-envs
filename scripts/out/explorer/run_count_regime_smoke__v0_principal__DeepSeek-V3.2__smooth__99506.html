import numpy as np
import pandas as pd

def model(regime, n, seed):
    # regime.config is a dict with "speed" key (float in [0.8,1.2])
    speed = regime.config.get("speed", 1.0)  # default if missing
    # Parameters from experimental fit
    alpha = 7.402
    beta = 3.669
    lam = alpha * (speed ** beta)
    # Ensure lam >= 0
    lam = max(lam, 0.0)
    # Set random seed
    rng = np.random.default_rng(seed)
    # Generate Poisson counts
    y = rng.poisson(lam, size=n).astype(float)
    # unit_id sequential from 0
    unit_id = np.arange(n, dtype=float)
    # Return DataFrame with columns exactly ["unit_id", "y"]
    return pd.DataFrame({"unit_id": unit_id, "y": y})
