import numpy as np
import pandas as pd

# Parameters estimated from data
a = 0.826
b = 5.546
noise_std = 0.6  # tuned to match residual variance ~6.2

def model(regime, n, seed):
    rng = np.random.default_rng(seed)
    speed = regime.config.get("speed", 1.0)
    repeats = regime.config.get("repeats_per_unit", 1)
    # Compute mean defect rate
    lam = a + b * speed
    if lam < 0:
        lam = 0.0
    # Generate n measurements
    pois = rng.poisson(lam, n)
    # Add small normal noise to break variance=mean and round
    noise = rng.normal(0, noise_std, n)
    y_raw = pois + noise
    y = np.round(y_raw).astype(int)
    y = np.clip(y, 0, None)  # no negative counts
    # Assign unit_id: groups of 'repeats' measurements
    n_lots = (n + repeats - 1) // repeats
    unit_ids = np.repeat(np.arange(n_lots), repeats)[:n]
    df = pd.DataFrame({
        "unit_id": unit_ids.astype(float),
        "y": y.astype(float)
    })
    return df
