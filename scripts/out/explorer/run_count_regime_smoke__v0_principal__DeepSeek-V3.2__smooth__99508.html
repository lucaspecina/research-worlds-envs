import numpy as np
import pandas as pd

# Fitted parameters from all data
LAMBDA0 = 7.286053989057371
BETA = 3.326700762697141

def model(regime, n, seed):
    """
    Generate n lots with defect counts y ~ Poisson(lambda(speed)).
    lambda(speed) = LAMBDA0 * exp(BETA * (speed - 1.0))
    regime.config["speed"] should be present.
    """
    # Set random seed for reproducibility
    rng = np.random.default_rng(seed)
    
    # Get speed from regime.config; if missing, default to 1.0 (though evaluation should provide)
    speed = regime.config.get("speed", 1.0)
    
    # Compute Poisson rate
    lam = LAMBDA0 * np.exp(BETA * (speed - 1.0))
    
    # Generate Poisson counts
    y = rng.poisson(lam, size=n)
    
    # Create DataFrame with unit_id sequential
    df = pd.DataFrame({
        'unit_id': np.arange(n, dtype=float),
        'y': y.astype(float)
    })
    return df
