import numpy as np
import pandas as pd

def model(regime, n, seed):
    """
    Generate n measurement rows of defects y.
    regime.config: speed (float), repeats_per_unit (int)
    """
    rng = np.random.default_rng(seed)
    speed = regime.config.get("speed", 1.0)
    repeats = regime.config.get("repeats_per_unit", 1)
    # Base parameters estimated from historical data at speed=1.0
    base_mean = 6.15
    base_theta = 3.228
    k = 1.905
    # Scale theta linearly with speed
    theta = base_theta * speed
    # Number of lots needed: ceil(n / repeats)
    n_lots = (n + repeats - 1) // repeats
    # Draw lot defect rates λ ~ Gamma(shape=k, scale=theta)
    lambdas = rng.gamma(shape=k, scale=theta, size=n_lots)
    # For each lot, generate repeats measurements
    data = []
    for i, lam in enumerate(lambdas):
        # Number of measurements for this lot: repeats, except maybe last lot
        n_this = repeats if (i < n_lots - 1) else (n - i * repeats)
        ys = rng.poisson(lam=lam, size=n_this)
        for y in ys:
            data.append({"unit_id": float(i), "y": float(y)})
    df = pd.DataFrame(data, columns=["unit_id", "y"])
    return df
