# Fit two-component Poisson mixture to historical data (speed=1.0)
from scipy.stats import poisson
import numpy as np

y_vals = df_full['y'].values
# Initial guesses
p1 = 0.5
lambda1 = 3.0
lambda2 = 10.0
max_iter = 100
tol = 1e-4
log_likelihood = -np.inf
for it in range(max_iter):
    # E-step: compute responsibilities
    prob1 = p1 * poisson.pmf(y_vals, lambda1)
    prob2 = (1 - p1) * poisson.pmf(y_vals, lambda2)
    total = prob1 + prob2
    r1 = prob1 / total
    r2 = prob2 / total
    # M-step: update parameters
    p1_new = r1.mean()
    lambda1_new = np.average(y_vals, weights=r1)
    lambda2_new = np.average(y_vals, weights=r2)
    # Check convergence
    new_log_likelihood = np.log(total).sum()
    if np.abs(new_log_likelihood - log_likelihood) < tol:
        break
    p1, lambda1, lambda2 = p1_new, lambda1_new, lambda2_new
    log_likelihood = new_log_likelihood

import pandas as pd
import numpy as np

def model(regime, n, seed):
    # Set random seed for reproducibility
    rng = np.random.default_rng(seed)
    # Extract parameters
    speed = regime.config.get("speed", 1.0)
    repeats = regime.config.get("repeats_per_unit", 1)
    # Mixture parameters (fitted from historical data at speed=1.0)
    p = 0.461  # probability of group 1 (low rate)
    lambda1 = 1.939
    lambda2 = 10.702
    # Scale by speed
    rate1 = lambda1 * speed
    rate2 = lambda2 * speed
    # Determine number of lots (ceil division)
    n_lots = (n + repeats - 1) // repeats
    # Generate group for each lot
    group = rng.choice([0, 1], size=n_lots, p=[p, 1-p])  # 0 for group1, 1 for group2
    # Generate defect counts for each measurement
    unit_ids = []
    ys = []
    for i in range(n_lots):
        g = group[i]
        rate = rate1 if g == 0 else rate2
        # Number of measurements for this lot: repeats, except maybe last lot
        r = repeats if (i+1)*repeats <= n else n - i*repeats
        # Generate r Poisson draws
        y_lot = rng.poisson(rate, size=r)
        # Assign unit_id (could be any identifier; we'll use i)
        unit_ids.extend([float(i)] * r)
        ys.extend(y_lot)
    # Create DataFrame
    df = pd.DataFrame({"unit_id": unit_ids, "y": ys})
    return df
