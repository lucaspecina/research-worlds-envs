import pandas as pd

# Fit linear relationships for k and theta vs speed (using three points)
speeds = np.array([0.8, 1.0, 1.2])
k_vals = np.array([1.992, 1.733, 1.412])
theta_vals = np.array([2.942, 3.727, 4.583])
# Linear regression
coeff_k = np.polyfit(speeds, k_vals, 1)
coeff_theta = np.polyfit(speeds, theta_vals, 1)
print("k = {}*speed + {}".format(coeff_k[0], coeff_k[1]))
print("theta = {}*speed + {}".format(coeff_theta[0], coeff_theta[1]))
# Define model function
def model(regime, n, seed):
    # Set random seed for reproducibility
    rng = np.random.default_rng(seed)
    speed = regime.config.get("speed", 1.0)
    repeats = regime.config.get("repeats_per_unit", 1)
    # Compute Gamma parameters
    k = coeff_k[0] * speed + coeff_k[1]
    theta = coeff_theta[0] * speed + coeff_theta[1]
    # Ensure positive
    k = max(k, 0.1)
    theta = max(theta, 0.1)
    # Determine number of lots needed
    if repeats <= 0:
        repeats = 1
    n_lots = (n + repeats - 1) // repeats  # ceiling division
    # Generate lot lambdas
    lambdas = rng.gamma(shape=k, scale=theta, size=n_lots)
    # Generate measurements
    unit_ids = []
    ys = []

        reps = repeats if i < n_lots - 1 else n - i * repeats
        unit_ids.extend([float(i)] * reps)
        ys.extend(rng.poisson(lam=lam, size=reps))
    # Truncate to n (should be exact)
    df = pd.DataFrame({
        'unit_id': unit_ids[:n],
        'y': ys[:n]
    })
    return df
