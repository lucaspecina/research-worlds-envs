import numpy as np
import pandas as pd

def model(regime, n, seed):
    rng = np.random.default_rng(seed)

    speed = float(regime.config.get("speed", 1.0))
    R = int(regime.config.get("repeats_per_unit", 1))
    if R < 1:
        R = 1

    # Calibrated from observational + experimental evidence
    mu0 = 6.21
    beta = 1.06
    k = 2.1  # gamma shape; induces NB-like overdispersion

    mu = mu0 * (speed ** beta)

    n_units = (n + R - 1) // R
    # Gamma-Poisson hierarchy: lot-specific latent rate
    rates = rng.gamma(shape=k, scale=mu / k, size=n_units)

    unit_ids = np.repeat(np.arange(n_units, dtype=int), R)[:n]

    idx = 0
    for i in range(n_units):
        m = min(R, n - idx)

    return pd.DataFrame({"unit_id": unit_ids, "y": y})
