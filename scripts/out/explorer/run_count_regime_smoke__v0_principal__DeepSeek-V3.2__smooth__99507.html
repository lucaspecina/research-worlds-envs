import numpy as np
import pandas as pd
from scipy.optimize import minimize

# Combine all data we have: historical (speed=1.0), low (0.8), high (1.2)
# Create arrays of speed and y
speeds = np.concatenate([
    np.full(len(df_hist_full), 1.0),
    np.full(len(df_exp_low), 0.8),
    np.full(len(df_exp_high), 1.2)
])
ys = np.concatenate([
    df_hist_full['y'].values,
    df_exp_low['y'].values,
    df_exp_high['y'].values
])
print("Total data points:", len(speeds))
print("Speeds unique:", np.unique(speeds))

# Define negative log-likelihood for Poisson with lambda = a * speed^b
def nll(params):
    a, b = params
    lam = a * (speeds ** b)
    # Poisson log-likelihood: y*log(lam) - lam - lgamma(y+1)
    # We ignore constant term
    ll = ys * np.log(lam) - lam
    return -ll.sum()

# Initial guess: a = mean at speed=1.0 = 7.168, b = 3.5
initial = [7.168, 3.5]
res = minimize(nll, initial, bounds=((1e-5, None), (0.1, 10)))
a_hat, b_hat = res.x
print(f"Fitted parameters: a = {a_hat:.4f}, b = {b_hat:.4f}")
print(f"Negative log-likelihood at optimum: {res.fun:.2f}")

# Compute predicted means at each speed
for s in [0.8, 1.0, 1.2]:
    lam = a_hat * (s ** b_hat)
    actual_mean = ys[speeds == s].mean()
    print(f"Speed {s}: predicted λ = {lam:.3f}, actual mean = {actual_mean:.3f}")

# Check if we need to account for underdispersion: compute Pearson chi-squared statistic
lam_all = a_hat * (speeds ** b_hat)
pearson = ((ys - lam_all)**2 / lam_all).sum()
print(f"Pearson chi-squared: {pearson:.2f}, degrees of freedom ~ {len(ys)-2}")
print(f"Mean of squared residuals: {np.mean((ys - lam_all)**2):.3f}")
# If Pearson >> df, indicates overdispersion; if << df, underdispersion.
# Let's compute variance-to-mean ratio per speed group
for s in [0.8, 1.0, 1.2]:
    y_group = ys[speeds == s]
    print(f"Speed {s}: mean {y_group.mean():.3f}, variance {y_group.var():.3f}, ratio {y_group.var()/y_group.mean():.3f}")
