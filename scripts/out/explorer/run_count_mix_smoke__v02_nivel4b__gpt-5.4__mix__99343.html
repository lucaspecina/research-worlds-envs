import numpy as np
import pandas as pd
import math

# Get a larger cheap baseline sample at speed=1
df_base = env.observe("archivo", 600)
print("BASE SHAPE:", df_base.shape)
print(df_base.head())

y = df_base["y"].astype(int).to_numpy()
n = len(y)
mu = y.mean()
var = y.var(ddof=1)
print("\nBaseline mean:", mu, "var:", var, "var/mean:", var/mu if mu > 0 else np.nan)

# Train/validation split
rng = np.random.default_rng(0)
idx = rng.permutation(n)
tr = y[idx[: int(0.7*n)]]
va = y[idx[int(0.7*n):]]

# Fit Poisson
mu_p = tr.mean()

# Fit NB2 via moments: Var = mu + mu^2/k
mu_nb = tr.mean()
var_tr = tr.var(ddof=1)
k_nb = np.inf if var_tr <= mu_nb else mu_nb**2 / (var_tr - mu_nb)

def poisson_logpmf_vec(y, lam):
    y = np.asarray(y)
    return y*np.log(lam) - lam - np.vectorize(math.lgamma)(y+1)

def nb_logpmf_vec(y, mu, k):
    # NB with mean mu and shape k; Var = mu + mu^2/k
    y = np.asarray(y)
    if np.isinf(k):
        return poisson_logpmf_vec(y, mu)
    p = k / (k + mu)
    return (np.vectorize(math.lgamma)(y + k)
            - math.lgamma(k)
            - np.vectorize(math.lgamma)(y + 1)
            + k*np.log(p)
            + y*np.log(1 - p))

ll_p = poisson_logpmf_vec(va, mu_p).mean()
ll_nb = nb_logpmf_vec(va, mu_nb, k_nb).mean()

print("\nCandidate family comparison on held-out archive rows:")
print({"poisson_mu": mu_p, "poisson_val_mean_loglik": ll_p})
print({"nb_mu": mu_nb, "nb_k": k_nb, "nb_val_mean_loglik": ll_nb})

import pandas as pd

# High-speed experiment mirroring the low-speed design
df_exp_hi = env.experiment(config={"speed": 1.2, "repeats_per_unit": 4}, n=120)
print("HIGH-SPEED EXPERIMENT SHAPE:", df_exp_hi.shape)

def summarize(df, label):
    g = df.groupby("unit_id")["y"].agg(["size","mean","var"]).reset_index()
    y = df["y"].to_numpy()
    overall_mean = float(np.mean(y))
    overall_var = float(np.var(y, ddof=1))
    within_vars = g["var"].dropna().to_numpy()
    mean_within_var = float(np.mean(within_vars))
    between_var_unit_means = float(np.var(g["mean"].to_numpy(), ddof=1))
    icc_like = between_var_unit_means / (between_var_unit_means + mean_within_var) if (between_var_unit_means + mean_within_var) > 0 else np.nan
    print(f"\n{label}")
    print({"overall_mean": overall_mean,
           "overall_var": overall_var,
           "var_over_mean": overall_var/overall_mean if overall_mean > 0 else np.nan,
           "mean_within_unit_var": mean_within_var,
           "between_var_unit_means": between_var_unit_means,
           "icc_like": icc_like})
    print("unit-level mean summary:")
    print(g["mean"].describe())

summarize(df_exp, "LOW speed=0.8")
summarize(df_exp_hi, "HIGH speed=1.2")

# Compare speed effect on means

mean_hi = df_exp_hi["y"].mean()
print("\nSpeed effect on mean:")

       "mean_1.0_archive": 6.465,
       "mean_1.2": float(mean_hi),

       "ratio_1.2_to_1.0": float(mean_hi/6.465),

# Estimate power-law exponent mu(speed)=mu1*speed^beta from the three means
speeds = np.array([0.8, 1.0, 1.2])

beta = np.polyfit(np.log(speeds), np.log(means), 1)[0]
print("\nPower-law beta estimate from means:", float(beta))
