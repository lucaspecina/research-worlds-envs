At this point I have a workable picture: counts are nonnegative integers with mean around 6 at speed 1, variance close to the mean, and repeated measurements within lot show only mild correlation at most. The safest compact model is therefore Poisson measurements with a smooth speed-dependent mean, and to honor repeated rows I should simply group rows by unit_id without imposing strong within-lot dependence. code = r'''
import numpy as np
import pandas as pd

def model(regime, n, seed):
    rng = np.random.default_rng(seed)
    speed = float(regime.config.get("speed", 1.0))
    R = int(regime.config.get("repeats_per_unit", 1))
    if R < 1:
        R = 1

    # Poisson defect counts with log-linear speed effect.
    # Calibrated from archive at speed=1.0 (~6.34 mean) and experiments showing
    # increasing rate with speed; weak within-unit dependence suggests treating
    # repeated measurements as approximately independent draws.
    base_mean = 6.34
    beta = 1.20
    lam = base_mean * np.exp(beta * (speed - 1.0))
    lam = max(lam, 1e-9)

    unit_ids = np.arange(n) // R
    y = rng.poisson(lam, size=n)

import numpy as np
import pandas as pd

def model(regime, n, seed):
    rng = np.random.default_rng(seed)
    speed = float(regime.config.get("speed", 1.0))
    R = int(regime.config.get("repeats_per_unit", 1))
    if R < 1:
        R = 1

    # Poisson defect counts with log-linear speed effect.
    # Calibrated from archive at speed=1.0 (~6.34 mean) and experiments showing
    # increasing rate with speed; weak within-unit dependence suggests treating
    # repeated measurements as approximately independent draws.
    base_mean = 6.34
    beta = 1.20
    lam = base_mean * np.exp(beta * (speed - 1.0))
    lam = max(lam, 1e-9)

    unit_ids = np.arange(n) // R
    y = rng.poisson(lam, size=n)

    return pd.DataFrame({
        "unit_id": unit_ids.astype(float),
        "y": y.astype(int)
    }, columns=["unit_id", "y"])
