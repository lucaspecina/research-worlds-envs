# Fit 2-component Poisson mixture with EM, using scipy for Poisson PMF
import numpy as np
from scipy.stats import poisson

y = df_all['y'].values.astype(float)
K = 2
n = len(y)

# Initialize
np.random.seed(42)
lambda_vals = np.random.uniform(1, 10, K)
weights = np.ones(K) / K

max_iter = 200
tol = 1e-6
log_likelihood_old = -np.inf

for it in range(max_iter):
    # E-step: compute responsibilities using log probabilities
    log_likelihoods = np.zeros((n, K))
    for k in range(K):
        log_likelihoods[:, k] = poisson.logpmf(y, lambda_vals[k])
    weighted_log_likelihoods = log_likelihoods + np.log(weights)
    # normalize
    max_log = weighted_log_likelihoods.max(axis=1, keepdims=True)
    exp_log = np.exp(weighted_log_likelihoods - max_log)
    resp = exp_log / exp_log.sum(axis=1, keepdims=True)
    
    # M-step
    Nk = resp.sum(axis=0)
    weights = Nk / n
    for k in range(K):
        lambda_vals[k] = (resp[:, k] * y).sum() / Nk[k]
    
    # Compute log-likelihood
    log_likelihood = np.log(np.exp(weighted_log_likelihoods - max_log).sum(axis=1) + 1e-16).sum() + max_log.sum()
    if np.abs(log_likelihood - log_likelihood_old) < tol:
        break
    log_likelihood_old = log_likelihood

print("Fitted mixture weights:", weights)
print("Fitted lambdas:", lambda_vals)
print("Iterations:", it+1)
print("Log-likelihood:", log_likelihood)

# Check BIC maybe
bic = -2 * log_likelihood + (2 * K - 1) * np.log(n)
print("BIC:", bic)

# Simulate from the mixture to compare distribution
sim_y = np.zeros_like(y)
for i in range(n):
    comp = np.random.choice(K, p=weights)
    sim_y[i] = np.random.poisson(lambda_vals[comp])
print("Simulated mean, std:", sim_y.mean(), sim_y.std())
print("Actual mean, std:", y.mean(), y.std())

def fit_poisson_mixture(y, K=2, max_iter=200):
    """Return weights, lambdas, loglik"""
    n = len(y)
    # Initialize
    np.random.seed(123)
    lambdas = np.random.uniform(1, 10, K)
    weights = np.ones(K) / K
    log_likelihood_old = -np.inf
    for it in range(max_iter):
        # E-step
        log_likelihoods = np.zeros((n, K))
        for k in range(K):
            log_likelihoods[:, k] = poisson.logpmf(y, lambdas[k])
        weighted_log_likelihoods = log_likelihoods + np.log(weights)
        max_log = weighted_log_likelihoods.max(axis=1, keepdims=True)
        exp_log = np.exp(weighted_log_likelihoods - max_log)
        resp = exp_log / exp_log.sum(axis=1, keepdims=True)
        # M-step
        Nk = resp.sum(axis=0)
        weights = Nk / n
        for k in range(K):
            lambdas[k] = (resp[:, k] * y).sum() / Nk[k]
        log_likelihood = np.log(np.exp(weighted_log_likelihoods - max_log).sum(axis=1) + 1e-16).sum() + max_log.sum()
        if np.abs(log_likelihood - log_likelihood_old) < 1e-6:
            break
        log_likelihood_old = log_likelihood
    return weights, lambdas, log_likelihood

# Moment matching for 2-component Poisson mixture
def solve_mixture(mean, var):
    # Solve for λ1, λ2, π such that λ1 <= λ2, 0<π<1
    # Use approximation: set λ1 = mean - delta, λ2 = mean + delta, solve for delta and π
    # Variance = mean + π(1-π)(λ2-λ1)^2
    # Let d = λ2 - λ1 = 2*delta
    # Then var - mean = π(1-π) d^2
    # Choose π=0.5 for simplicity, then d = sqrt(4*(var - mean))
    # But var - mean could be negative -> not possible with mixture (mixture always increases variance)
    if var <= mean:
        # No separation needed, single Poisson
        return None, None, None
    # Choose π=0.5 (equal weights)
    pi = 0.5
    d = np.sqrt((var - mean) / (pi * (1 - pi)))
    lambda1 = mean - d/2
    lambda2 = mean + d/2
    if lambda1 < 0:
        lambda1 = 0.1
        lambda2 = (mean - pi*lambda1) / (1 - pi)  # adjust
    return pi, lambda1, lambda2

def simulate_model(speed, repeats_per_unit, n_measurements, seed):
    np.random.seed(seed)
    a = 6.0
    c = 1.3
    p0 = 1.0
    p1 = -0.7
    pi_low = p0 + p1*(speed - 1)
    pi_low = max(0, min(1, pi_low))
    # Determine number of lots
    n_lots = int(np.ceil(n_measurements / repeats_per_unit))
    # Generate lot-level component and rate
    lot_component = np.random.binomial(1, 1 - pi_low, size=n_lots)  # 1 for high, 0 for low
    base_rates = a * speed * (1 + (c - 1) * lot_component)  # low: a*speed, high: a*speed*c
    # Generate measurements
    measurements = []
    unit_ids = []
    for i in range(n_lots):
        rate = base_rates[i]
        repeats = repeats_per_unit if i < n_lots - 1 else n_measurements - i * repeats_per_unit
        y_vals = np.random.poisson(rate, size=repeats)
        measurements.extend(y_vals)
        unit_ids.extend([float(i)] * repeats)
    df = pd.DataFrame({'unit_id': unit_ids, 'y': measurements})
    return df
