import numpy as np
import pandas as pd

def model(regime, n, seed):
    rng = np.random.default_rng(seed)

    speed = float(regime.config.get("speed", 1.0))
    R = int(regime.config.get("repeats_per_unit", 1))
    R = max(1, min(4, R))

    # Baseline latent lot-type mixture at speed=1.0.
    # Chosen to reproduce archive mean/variance and the apparent low/mid/high lot clusters.
    weights = np.array([0.40, 0.20, 0.40], dtype=float)
    base_rates = np.array([1.6, 5.8, 10.2], dtype=float)

    # Speed multiplier: monotone and convex, fitted from the extreme-speed experiments.
    # Anchored so multiplier(1.0)=1.
    alpha = 1.85
    mult = speed ** alpha

    n_units = (n + R - 1) // R
    unit_ids = np.repeat(np.arange(n_units), R)[:n]

    types = rng.choice(len(weights), size=n_units, p=weights)
    lam_units = base_rates[types] * mult

    y = np.empty(n, dtype=int)
    idx = 0
    for u in range(n_units):
        reps = min(R, n - idx)
        y[idx:idx+reps] = rng.poisson(lam_units[u], size=reps)
        idx += reps

    return pd.DataFrame({
        "unit_id": unit_ids.astype(float),
        "y": y.astype(float),
    })
