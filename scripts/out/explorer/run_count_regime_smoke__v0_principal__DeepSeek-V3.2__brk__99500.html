    # For speed >= 1.2, use linear extrapolation using last two points
    # In between, linear interpolation
    if speed <= speed_points[0]:
        lam = mean_points[0]
    elif speed >= speed_points[-1]:

        slope = (mean_points[-1] - mean_points[-2]) / (speed_points[-1] - speed_points[-2])
