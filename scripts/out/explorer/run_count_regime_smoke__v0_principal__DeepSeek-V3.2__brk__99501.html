# Estimate negative binomial dispersion parameter alpha (for var = mean + alpha*mean^2)
# Using method of moments: alpha = (variance - mean) / mean^2
# Compute for each speed group and take average
alphas = []
for s, m, v in zip(speeds, means, [3.639, 5.577, 4.319, 14.213, 20.493]):
    if m > 0:
        alpha = (v - m) / (m**2)
        alphas.append(alpha)
alpha_mean = np.mean(alphas)
print(f"Alpha estimates per speed: {alphas}")
print(f"Mean alpha: {alpha_mean:.4f}")

# If alpha is negative (underdispersion), fall back to Poisson (alpha=0)
if alpha_mean < 0:
    alpha_mean = 0.0
    print("Underdispersion detected, using Poisson (alpha=0).")

# Define the generative model
def model(regime, n, seed):
    """
    regime.config: dict with 'speed' key (float in [0.8, 1.2])
    regime.context: empty dict
    Returns DataFrame with columns ['unit_id', 'y']
    """
    rng = np.random.default_rng(seed)
    speed = regime.config.get("speed", 1.0)  # default if not provided (though evaluation will provide)
    # Compute log mean
    log_mu = a + b * speed + c * speed * speed
    mu = np.exp(log_mu)
    # If alpha_mean is zero, use Poisson; else negative binomial
    if alpha_mean <= 1e-10:
        y = rng.poisson(mu, n)
    else:
        # NB2 parameterization: mean = mu, variance = mu + alpha * mu^2
        # p = 1 / (1 + alpha * mu)   # success probability
        # r = mu * p / (1 - p)       # number of successes
        p = 1.0 / (1.0 + alpha_mean * mu)
        r = mu * p / (1.0 - p)
        y = rng.negative_binomial(r, p, n)
    # Ensure integer and non-negative (already is)
    y = y.astype(int)
    # Create unit_id sequential from 0 to n-1
    unit_id = np.arange(n)
    df = pd.DataFrame({'unit_id': unit_id, 'y': y})
    return df
