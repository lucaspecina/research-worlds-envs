code = r'''
import numpy as np
import pandas as pd

def model(regime, n, seed):
    rng = np.random.default_rng(seed)

    speed = float(regime.config.get("speed", 1.0))
    repeats = int(regime.config.get("repeats_per_unit", 1))

    # Baseline from archive / experiments:
    # mean at speed 1.0 around 6.8-7.1, overdispersion consistent with NB size ~2.5
    base_mean = 6.9
    nb_k = 2.5  # gamma-poisson shape

    # From experiments, mean appears highest around speed=1.0 and lower at both 0.8 and 1.2.
    # Use a symmetric concave quadratic over the allowed range.
    # Targets roughly: mu(1.0)=6.9, mu(0.8)≈5.6, mu(1.2)≈6.2
    d = speed - 1.0
    mu = base_mean * max(0.05, 1.0 - 0.8 * d - 2.2 * d * d)

    # Number of lots implied by measurement rows and repeats per lot
    n_units = (n + repeats - 1) // repeats
    unit_ids = np.repeat(np.arange(n_units), repeats)[:n]

    # Gamma-Poisson mixture:
    # lambda_j ~ Gamma(shape=k, scale=mu/k), y_ij ~ Poisson(lambda_j)
    shape = nb_k
    scale = mu / shape
    lam = rng.gamma(shape=shape, scale=scale, size=n_units)
    y = rng.poisson(lam=np.repeat(lam, repeats)[:n]).astype(int)

    return pd.DataFrame({
        "unit_id": unit_ids.astype(float),
