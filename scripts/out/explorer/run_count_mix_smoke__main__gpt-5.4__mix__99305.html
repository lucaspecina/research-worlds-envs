all_df = pd.concat([obs, e1, e2, e3], ignore_index=True)
print("all data shape:", all_df.shape)

# Speed-wise row moments
summ = all_df.groupby("speed")["y"].agg(["mean","var","count"])
print("\nrow-level by speed:")
print(summ)

# Estimate gamma shape alpha from repeated-measures experiments:
# For Poisson-Gamma with repeats m, Cov(within same unit)=Var(lambda)=mu^2/alpha, E[within var]=E[lambda]=mu
# Use ANOVA-ish decomposition on unit means.
for sp, d in [("0.8", exp_low), ("1.0", exp_mid), ("1.2", exp1)]:
    grp = d.groupby("unit_id")["y"]
    unit_means = grp.mean()
    mu = d["y"].mean()
    between = unit_means.var()
    alpha = mu**2 / max(between - mu/4.0, 1e-9)  # subtract sampling noise of mean with m=4
    print(f"\nspeed {sp}: mu={mu:.3f}, unit_mean_var={between:.3f}, alpha_est={alpha:.3f}")

# Fit simple mean(speed) models using the three experimental means
speeds = np.array([0.8, 1.0, 1.2])
mus = np.array([exp_low['y'].mean(), exp_mid['y'].mean(), exp1['y'].mean()])
coef_lin = np.polyfit(speeds, mus, 1)
coef_log = np.polyfit(speeds, np.log(mus), 1)
print("\nlinear fit mu = a + b*speed:", coef_lin)
print("log-linear fit mu = exp(a + b*speed):", coef_log)
print("pred linear at speeds:", list(zip(speeds, np.polyval(coef_lin, speeds))))
print("pred loglinear at speeds:", list(zip(speeds, np.exp(np.polyval(coef_log, speeds)))))

Razonamiento

REASONING: The shape parameter is fairly stable around 1.8–2.2 across speeds, which strongly supports a Gamma-Poisson hierarchy with approximately constant Gamma shape and speed-dependent mean. Mean vs speed is smooth; linear and log-linear are both plausible, but log-linear guarantees positivity and often extrapolates better within a bounded range, so it's a safer generative choice. We have enough to submit: sample a lot-level rate lambda ~ Gamma(shape=alpha, scale=mu/alpha), then repeated measurements y ~ Poisson(lambda), with unit_ids grouped according to repeats_per_unit.

Código

code = r'''
import pandas as pd
import numpy as np

def model(regime, n, seed):
    rng = np.random.default_rng(seed)

    speed = float(regime.config.get("speed", 1.0))
    repeats = int(regime.config.get("repeats_per_unit", 1))
    repeats = max(1, repeats)

    # Mean defects as a smooth function of speed, fit from experiments.
    # log(mu) = a + b * speed
    a = 0.84916611
    b = 0.89427669
    mu = float(np.exp(a + b * speed))

    # Shared gamma shape capturing lot-to-lot heterogeneity.
    alpha = 1.9
    scale = mu / alpha

    n_units = (n + repeats - 1) // repeats
    lambdas = rng.gamma(shape=alpha, scale=scale, size=n_units)

    unit_ids = np.repeat(np.arange(n_units), repeats)[:n]
    y = np.empty(n, dtype=int)
    idx = 0
    for u in range(n_units):
        m = min(repeats, n - idx)
        y[idx:idx+m] = rng.poisson(lam=lambdas[u], size=m)
        idx += m

    return pd.DataFrame({"unit_id": unit_ids, "y": y})
'''
