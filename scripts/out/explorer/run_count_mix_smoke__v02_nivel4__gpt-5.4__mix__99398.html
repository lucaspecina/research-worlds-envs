import pandas as pd

# collect unit-level means from experiments (more informative about latent lot rates)
exp_base = exp1.copy()
datasets = {
    0.8: exp_lo.copy(),
    1.0: exp_base.copy(),
    1.2: exp_hi.copy(),
}

unit_means = []
for s, df in datasets.items():
    g = df.groupby("unit_id")["y"].mean().reset_index()
    g["speed"] = s
    unit_means.append(g[["speed", "y"]].rename(columns={"y":"unit_mean"}))
um = pd.concat(unit_means, ignore_index=True)
print("unit means by speed:")
print(um.groupby("speed")["unit_mean"].describe())

# fit beta from mean scaling
speeds = np.array(sorted(datasets.keys()))
means = np.array([datasets[s]["y"].mean() for s in speeds])
beta = np.polyfit(np.log(speeds), np.log(means), 1)[0]
print("\nmeans:", dict(zip(speeds, means)))
print("beta from log-log fit:", float(beta))

# rescale unit means back to baseline and fit 2-component Gaussian-ish proxy on unit means via EM
z = um["unit_mean"].to_numpy() / (um["speed"].to_numpy() ** beta)

def fit_two_normal_mix(x, iters=200, seed=0):
    rng = np.random.default_rng(seed)
    m1, m2 = np.quantile(x, [0.3, 0.7])
    s1 = s2 = np.std(x)
    w = 0.5
    for _ in range(iters):
        # normal pdfs
        p1 = w * np.exp(-0.5*((x-m1)/s1)**2) / max(s1,1e-6)
        p2 = (1-w) * np.exp(-0.5*((x-m2)/s2)**2) / max(s2,1e-6)
        r1 = p1 / np.maximum(p1+p2, 1e-12)
        w = r1.mean()
        m1 = (r1*x).sum()/np.maximum(r1.sum(),1e-12)
        m2 = (((1-r1)*x).sum())/np.maximum((1-r1).sum(),1e-12)
        s1 = np.sqrt((r1*(x-m1)**2).sum()/np.maximum(r1.sum(),1e-12))
        s2 = np.sqrt((((1-r1)*(x-m2)**2).sum())/np.maximum((1-r1).sum(),1e-12))
        s1 = max(s1, 0.2); s2 = max(s2, 0.2)
    if m1 > m2:
        w, m1, m2, s1, s2 = 1-w, m2, m1, s2, s1
    return w, m1, m2, s1, s2

w, l1, l2, s1, s2 = fit_two_normal_mix(z, seed=1)
print("\n2-cluster proxy on baseline-scaled unit means:")
print("w, l1, l2, s1, s2 =", w, l1, l2, s1, s2)

    cls = rng.random(n_units) < w
    base_rates = np.where(cls, lam1, lam2)
    rates = base_rates * (speed ** beta)

    y = rng.poisson(np.repeat(rates, R)[:n]).astype(int)
    return pd.DataFrame({"unit_id": unit_ids.astype(float), "y": y.astype(float)})
