# Fit negative binomial to historical y (speed=1.0)
from scipy.stats import nbinom
import numpy as np

y_hist = full_hist['y'].values
# Method of moments for negative binomial: mean = r*(1-p)/p, var = r*(1-p)/p^2
mean = y_hist.mean()
var = y_hist.var()
# Solve for r and p
p = mean / var  # if var > mean
r = mean * p / (1 - p)
print("Method of moments estimates for NB(r, p):")
print("r =", r)
print("p =", p)
# Alternatively, use MLE via optimization
import warnings
warnings.filterwarnings('ignore')
from scipy.optimize import minimize
def neg_loglik(params):
    r, p = params
    # p must be in (0,1), r > 0
    if p <= 0 or p >= 1 or r <= 0:
        return 1e10
    return -np.sum(nbinom.logpmf(y_hist, r, p))
res = minimize(neg_loglik, [r, p], bounds=[(1e-3, 100), (1e-3, 1-1e-3)])
if res.success:
    r_mle, p_mle = res.x
    print("MLE estimates:")
    print("r =", r_mle)
    print("p =", p_mle)
else:
    print("MLE failed")
# Check if mixture of two Poissons might be better? Compute histogram
hist_counts, bin_edges = np.histogram(y_hist, bins=np.arange(0, y_hist.max()+2)-0.5)
print("\nHistogram of y (counts):")
for i, count in enumerate(hist_counts):
