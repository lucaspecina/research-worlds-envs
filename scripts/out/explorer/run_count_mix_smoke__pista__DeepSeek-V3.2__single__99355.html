# Build model code
model_code = """
import numpy as np
import pandas as pd

def model(regime, n, seed):
    # Set random seed for reproducibility
    rng = np.random.default_rng(seed)
    
    # Extract configuration
    speed = regime.config.get("speed", 1.0)
    repeats = regime.config.get("repeats_per_unit", 1)
    
    # Base defect rate per unit speed (estimated from data)
    beta = 6.53
    
    # Poisson rate
    lam = beta * speed
    
    # Generate defect counts for n measurements
    y = rng.poisson(lam, size=n)
    
    # Assign unit_ids: groups of 'repeats' measurements share same unit_id
    # unit_id starts at 0
    n_lots = (n + repeats - 1) // repeats  # ceiling division
    unit_ids = []
    for i in range(n_lots):
        # number of measurements in this lot (last lot may be shorter)
        lot_size = repeats if i < n_lots - 1 else n - i * repeats
        unit_ids.extend([float(i)] * lot_size)
    
    # Create DataFrame
    df = pd.DataFrame({
        'unit_id': unit_ids,
        'y': y
    })
    return df
"""
