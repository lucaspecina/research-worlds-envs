from math import lgamma

# Fit 2- and 3-component Poisson mixtures by EM on this sample
counts = y.value_counts().sort_index()
xs = counts.index.to_numpy(dtype=float)
ws = counts.to_numpy(dtype=float)

def pois_pmf_vec(x, lam):
    lam = np.maximum(lam, 1e-9)
    return np.exp(x*np.log(lam) - lam - np.vectorize(lambda t: lgamma(t+1))(x))

def fit_mix(K, inits, n_iter=200):
    best = None
    for init_lams in inits:
        pis = np.ones(K) / K
        lams = np.array(init_lams, dtype=float)
        for _ in range(n_iter):
            comps = np.vstack([pis[k] * pois_pmf_vec(xs, lams[k]) for k in range(K)]).T
            resp = comps / (comps.sum(axis=1, keepdims=True) + 1e-300)
            Nk = (ws[:, None] * resp).sum(axis=0)
            pis = Nk / ws.sum()
            lams = ((ws[:, None] * resp) * xs[:, None]).sum(axis=0) / np.maximum(Nk, 1e-12)
            lams = np.maximum(lams, 1e-6)
        comps = np.vstack([pis[k] * pois_pmf_vec(xs, lams[k]) for k in range(K)]).T
        mix = comps.sum(axis=1)
        ll = np.sum(ws * np.log(mix + 1e-300))
        out = (ll, pis.copy(), lams.copy())
        if best is None or ll > best[0]:
            best = out
    return best

inits3 = [[1, 4, 10], [0.5, 3, 9], [1, 6, 12], [2, 5, 11]]

best3 = fit_mix(3, inits3)

import numpy as np
import pandas as pd

def model(regime, n, seed):
    rng = np.random.default_rng(seed)

    speed = float(regime.config.get("speed", 1.0))
    R = int(regime.config.get("repeats_per_unit", 1))
    if R < 1:
        R = 1

    # Baseline lot-type mixture at speed=1.0, estimated from archive.
    mix_p = np.array([0.5689456, 0.4310544], dtype=float)
    base_lam = np.array([1.88636913, 10.57119135], dtype=float)

    n_units = (n + R - 1) // R
    unit_ids = np.repeat(np.arange(n_units), R)[:n]

    # One latent type per unit/lot; repeated measurements are fresh counts for that lot.
    z = rng.choice(len(mix_p), size=n_units, p=mix_p)
    lam_units = base_lam[z] * speed

    y = np.empty(n, dtype=int)
    idx = 0
    for u in range(n_units):
        r_u = min(R, n - idx)
        y[idx:idx+r_u] = rng.poisson(lam_units[u], size=r_u)
        idx += r_u

    return pd.DataFrame({
        "unit_id": unit_ids.astype(float),
        "y": y.astype(float),
    })
