def simulate_nb(n, speed, repeats_per_unit, seed, theta, base_rate, power=1.0):
    """Negative binomial with mean = base_rate * speed**power, dispersion theta."""
    np.random.seed(seed)
    # Mean per measurement
    mu = base_rate * (speed ** power)
    # Negative binomial parameters: p = theta/(theta+mu), r = theta
    p = theta / (theta + mu)
    # Draw defects per measurement
    y_per = np.random.negative_binomial(theta, 1-p, size=n)
