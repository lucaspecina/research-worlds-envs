Código

import numpy as np, pandas as pd

speeds = np.array([0.8, 1.0, 1.1, 1.2], dtype=float)
means = np.array([4.2, 5.5633333333333335, 11.1375, 14.05], dtype=float)

# Candidate fits on mean scale
coef_lin = np.polyfit(speeds, means, 1)
coef_quad = np.polyfit(speeds, means, 2)

pred_lin = np.polyval(coef_lin, speeds)
pred_quad = np.polyval(coef_quad, speeds)

# Also power around speed: mean = a * speed^b
b_pow, loga_pow = np.polyfit(np.log(speeds), np.log(means), 1)
pred_pow = np.exp(loga_pow) * speeds**b_pow

print("observed means:")
print(pd.DataFrame({"speed": speeds, "mean": means}))
print("\nlinear fit coeffs:", coef_lin)
print("linear preds:", np.round(pred_lin, 3), "sse:", float(np.sum((means - pred_lin)**2)))

print("\nquadratic fit coeffs:", coef_quad)
print("quadratic preds:", np.round(pred_quad, 3), "sse:", float(np.sum((means - pred_quad)**2)))

print("\npower fit a,b:", float(np.exp(loga_pow)), float(b_pow))
print("power preds:", np.round(pred_pow, 3), "sse:", float(np.sum((means - pred_pow)**2)))

# Also inspect positivity over full range
grid = np.linspace(0.8, 1.2, 9)
print("\nquadratic over grid:")
print(pd.DataFrame({"speed": grid, "lambda_quad": np.round(np.polyval(coef_quad, grid), 3)}))
