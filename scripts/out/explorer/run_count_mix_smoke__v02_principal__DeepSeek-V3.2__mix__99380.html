# Define and submit the model
model_code = """
import numpy as np
import pandas as pd

def model(regime, n, seed):
    # Set random seed for reproducibility
    rng = np.random.RandomState(seed)
    
    # Parameters from data fitting
    base_rate = 7.0   # defects per unit speed
    k = 3.0           # dispersion shape parameter
    
    speed = regime.config.get('speed', 1.0)
    repeats = regime.config.get('repeats_per_unit', 1)
    
    # Mean defect rate for given speed
    mu = base_rate * speed
    
    # Determine number of lots needed
    n_lots = (n + repeats - 1) // repeats  # ceiling division
    # Generate lot-specific rates
    lambdas = rng.gamma(shape=k, scale=mu / k, size=n_lots)
    
    # Generate measurements
    y_list = []
    unit_id_list = []
    remaining = n
    for i, lam in enumerate(lambdas):
        reps = min(repeats, remaining)
        y_list.extend(rng.poisson(lam, size=reps))
        unit_id_list.extend([float(i)] * reps)
        remaining -= reps
        if remaining <= 0:
            break
    
    df = pd.DataFrame({
        'unit_id': unit_id_list,
        'y': y_list
    })
    return df
"""
