import numpy as np
import pandas as pd

def model(regime, n, seed):
    """
    Generative model for defect counts.
    regime.config: dict with keys 'speed' (float) and 'repeats_per_unit' (int).
    n: number of measurement rows.
    seed: for reproducibility.
    Returns DataFrame with columns ['unit_id', 'y'].
    """
    rng = np.random.default_rng(seed)
    speed = regime.config.get('speed', 1.0)
    repeats = int(regime.config.get('repeats_per_unit', 1))
    
    # Parameters estimated from historical data at speed=1.0
    mu0 = 6.33          # mean defects at speed=1.0
    var0 = 24.3         # variance at speed=1.0
    theta0 = var0 / mu0   # scale of Gamma at speed=1.0
    k = mu0 / theta0      # shape of Gamma (constant across speeds)
    
    # Scale the Gamma distribution: mean = mu0 * speed, variance = k * (theta0 * speed)^2
    scale = theta0 * speed
    
    # Number of lots needed (each lot will have up to 'repeats' measurements)
    n_lots = int(np.ceil(n / repeats))
    lambdas = rng.gamma(shape=k, scale=scale, size=n_lots)
    
    unit_ids = []
    ys = []
    for i, lam in enumerate(lambdas):
        for r in range(repeats):
            if len(ys) >= n:
                break
            y = rng.poisson(lam)
            unit_ids.append(float(i))
            ys.append(float(y))
    
    return pd.DataFrame({'unit_id': unit_ids, 'y': ys})
