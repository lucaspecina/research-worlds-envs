    # variance/mean = 1 + beta * speed => beta = (variance/mean - 1) / speed
    # At speed=1.0: beta_est = (24.474/6.21 - 1) / 1.0 = 3.941 - 1 = 2.941
    # alpha_est = mean / (beta * speed) = 6.21 / (2.941 * 1) = 2.111

    scale = beta * speed
