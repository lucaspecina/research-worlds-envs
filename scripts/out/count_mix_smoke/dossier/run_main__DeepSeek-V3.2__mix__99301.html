import numpy as np
import pandas as pd

def model(regime, n, seed):
    """
    Generate n measurements of defect counts y.
    regime.config: dict with keys "speed" (float in [0.8,1.2]) and "repeats_per_unit" (int in [1,4])
    regime.context: empty dict.
    Returns DataFrame with columns ['unit_id', 'y'].
    """
    rng = np.random.default_rng(seed)
    speed = regime.config.get("speed", 1.0)
    R = regime.config.get("repeats_per_unit", 1)
    # Fixed parameters estimated from data
    k = 1.8448   # Gamma shape
    theta0 = 3.4317  # Gamma scale
    
    # Number of lots needed: ceil(n / R)
    n_lots = (n + R - 1) // R
    # Draw lot defect rates
    theta = rng.gamma(shape=k, scale=theta0, size=n_lots)
    lambda_i = speed * theta  # lot-specific Poisson rate
    
    # Generate measurements
    unit_ids = []
    y_vals = []
    for i in range(n_lots):
        # number of measurements for this lot: R except possibly last lot
        reps = R if i < n_lots - 1 else n - i * R
        y_lot = rng.poisson(lambda_i[i], size=reps)
        unit_ids.extend([float(i)] * reps)
        y_vals.extend(y_lot)
    
    df = pd.DataFrame({'unit_id': unit_ids, 'y': y_vals})
    return df
