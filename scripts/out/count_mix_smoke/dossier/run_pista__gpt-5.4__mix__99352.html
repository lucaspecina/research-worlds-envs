# Estimate gamma-poisson shape from repeated experiments:
# For Poisson-Gamma with unit-specific lambda:
# E[within-unit var] ~= E[lambda] = m
# Var(unit means) = Var(lambda) + E[lambda]/R
# so Var(lambda) ~= Var(unit means) - m/R
def est_from_rep(data, speed, R=4):
    g = data.groupby("unit_id")["y"]
    unit_means = g.mean()
    within_vars = g.var()
    m = data["y"].mean()
    vlam = max(unit_means.var() - m / R, 1e-9)
    k = (m ** 2) / vlam
    return {"speed": speed, "mean": m, "k": k, "vlam": vlam, "within_mean_var": within_vars.mean(), "var_unit_means": unit_means.var()}
