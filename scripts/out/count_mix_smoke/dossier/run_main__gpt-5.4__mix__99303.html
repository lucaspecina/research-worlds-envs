Razonamiento

REASONING: The multi-speed results are noisy because each speed only has 30 lots, and 1.2 oddly came in below 1.1, but the broad pattern is still increasing mean with speed. Budget is essentially exhausted, so the pragmatic move is to fit a simple hierarchical count model from what we have: lot-specific Gamma rates with Poisson repeated measurements, using a linear mean function in speed centered at 1.0 and a constant Gamma shape inferred from the strong overdispersion at speed 1.0. That should capture both the marginal negative-binomial behavior and the within-unit correlation. I’ll estimate baseline mean from all speed-1 data, estimate the slope from the four experimental means plus speed-1 anchor, estimate shape from speed-1 variance, then submit.

Código

import numpy as np
import pandas as pd

# Gather speed-1 info from archive + experiment
mean1_arch = df["y"].mean()
var1_arch = df["y"].var()
mean1_exp = exp1["y"].mean()
var1_exp = exp1["y"].var()

mu1 = np.mean([mean1_arch, mean1_exp])
var1 = np.mean([var1_arch, var1_exp])

# Negative-binomial/Poisson-gamma moment match at speed 1:
# Var = mu + mu^2 / k  => k = mu^2 / (Var - mu)
k = mu1**2 / max(var1 - mu1, 1e-6)

# Fit linear mean(speed) using experimental means plus speed 1 anchor
speed_pts = np.array([0.8, 0.9, 1.0, 1.1, 1.2], dtype=float)
mean_pts = np.array([
    exps[0.8]["y"].mean(),
    exps[0.9]["y"].mean(),
    mu1,
    exps[1.1]["y"].mean(),
    exps[1.2]["y"].mean()
], dtype=float)
coef = np.polyfit(speed_pts, mean_pts, 1)
b1, b0 = coef[0], coef[1]

print("Estimated mu1, var1, k:", mu1, var1, k)
print("Linear mean coefficients slope/intercept:", b1, b0)
print("Means by speed points:", list(zip(speed_pts, mean_pts)))

code = f'''
import numpy as np
import pandas as pd

MU_INTERCEPT = {float(b0)}
MU_SLOPE = {float(b1)}
GAMMA_SHAPE = {float(k)}

def model(regime, n, seed):
    rng = np.random.default_rng(seed)
    speed = float(regime.config.get("speed", 1.0))
    repeats = int(regime.config.get("repeats_per_unit", 1))
    repeats = max(1, min(4, repeats))
    mu = MU_INTERCEPT + MU_SLOPE * speed
    mu = max(1e-6, mu)

    n_units = (n + repeats - 1) // repeats
    # Poisson-Gamma mixture: lambda_u ~ Gamma(shape=k, scale=mu/k)
    k = max(1e-6, GAMMA_SHAPE)
    scale = mu / k
    lam = rng.gamma(shape=k, scale=scale, size=n_units)

    unit_ids = np.repeat(np.arange(n_units), repeats)[:n]
    y = np.empty(n, dtype=int)
    idx = 0
    for u in range(n_units):
        r = min(repeats, n - idx)
        y[idx:idx+r] = rng.poisson(lam[u], size=r)
        idx += r

    return pd.DataFrame({{"unit_id": unit_ids.astype(float), "y": y.astype(float)}})
'''
