def simulate_speed(speed, n_measurements, repeats_per_unit, seed):
    rng = np.random.default_rng(seed)
    mu = base * speed ** gamma
    scale = mu / alpha
    # Number of lots needed
    n_lots = int(np.ceil(n_measurements / repeats_per_unit))
    # Generate lot rates
    lot_rates = rng.gamma(shape=alpha, scale=scale, size=n_lots)
    # Expand to measurements per lot
    lot_ids = np.repeat(np.arange(n_lots), repeats_per_unit)
    # Truncate if more than n_measurements
    if len(lot_ids) > n_measurements:
        lot_ids = lot_ids[:n_measurements]
    # Generate y
    y = rng.poisson(lot_rates[lot_ids])
    # Build DataFrame
    df = pd.DataFrame({
        'unit_id': lot_ids.astype(float),
        'y': y.astype(float)
    })
    return df
