REASONING: This rules out “repeat is an identical copy of lot defects”; repeated measurements are noisy draws, but likely correlated within lot because all 4 readings on a unit tend to be in a similar range. A natural model is hierarchical Poisson-gamma: each lot has latent defect intensity λ depending on speed, and repeated measurements are Poisson(λ). Marginally that gives negative-binomial overdispersion, and within-unit correlation. code = r'''
import numpy as np
import pandas as pd

def model(regime, n, seed):
    rng = np.random.default_rng(seed)

    speed = float(regime.config.get("speed", 1.0))
    R = int(regime.config.get("repeats_per_unit", 1))
    R = max(1, min(4, R))

    # Mean defect rate as a function of speed.
    # Calibrated from experiments at speeds 0.8, 1.0, 1.2.
    mu = -3.7 + 11.0 * speed
    if mu <= 1e-6:
        mu = 1e-6

    # Lot-to-lot heterogeneity (Poisson-Gamma => Negative Binomial marginally)
    k = 3.2  # pooled from repeated-measure experiments

    n_units = (n + R - 1) // R
    unit_ids = np.repeat(np.arange(n_units), R)[:n]

    lam = rng.gamma(shape=k, scale=mu / k, size=n_units)
    y = rng.poisson(lam=np.repeat(lam, R)[:n]).astype(int)

    return pd.DataFrame({"unit_id": unit_ids.astype(float), "y": y.astype(float)})
'''

    R = int(regime.config.get("repeats_per_unit", 1))
    R = max(1, min(4, R))

    # Mean defect rate as a function of speed.
    # Calibrated from experiments at speeds 0.8, 1.0, 1.2.
    mu = -3.7 + 11.0 * speed
    if mu <= 1e-6:
        mu = 1e-6

    # Lot-to-lot heterogeneity (Poisson-Gamma => Negative Binomial marginally)
    k = 3.2  # pooled from repeated-measure experiments

    n_units = (n + R - 1) // R
    unit_ids = np.repeat(np.arange(n_units), R)[:n]

    lam = rng.gamma(shape=k, scale=mu / k, size=n_units)
    y = rng.poisson(lam=np.repeat(lam, R)[:n]).astype(int)

    return pd.DataFrame({"unit_id": unit_ids.astype(float), "y": y.astype(float)})
