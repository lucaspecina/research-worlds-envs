import pandas as pd
import numpy as np

# Mock regime for testing
class MockRegime:
    def __init__(self, speed=1.0, repeats=1):
        self.config = {"speed": speed, "repeats_per_unit": repeats}
        self.context = {}

def model(regime, n, seed):
    np.random.seed(seed)
    # Parameters estimated from historical data
    baseline_mean = 6.41
    r = 2.155
    speed = regime.config.get("speed", 1.0)
    repeats = regime.config.get("repeats_per_unit", 1)
    mean = baseline_mean * speed
    shape = r
    scale = mean / r  # Gamma scale parameter
    # Number of lots needed
    n_lots = (n + repeats - 1) // repeats  # ceil division
    # Draw lambda for each lot
    lambdas = np.random.gamma(shape=shape, scale=scale, size=n_lots)
    # Generate measurements
    unit_ids = []
    ys = []
    for i, lam in enumerate(lambdas):
        # Number of measurements for this lot: repeats, except possibly last lot
        reps = repeats if (i < n_lots - 1) else (n - i * repeats)
        for _ in range(reps):
            unit_ids.append(float(i))
            ys.append(np.random.poisson(lam))
    df = pd.DataFrame({"unit_id": unit_ids, "y": ys})
    return df
