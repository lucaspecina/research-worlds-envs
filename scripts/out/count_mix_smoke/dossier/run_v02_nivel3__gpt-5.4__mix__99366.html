Razonamiento

REASONING: This strongly refutes the idea that repeats are identical measurements of a fixed lot-level defect count; instead, repeated measurements within a unit are variable. But the unit means seem to cluster into low and high levels, so a natural model is latent lot type determining a per-measurement Poisson rate, with repeated measurements conditionally independent given type. To separate “few discrete types” from continuous random effects, I want a cheap diagnostic on the experiment data: compare between-unit variance of unit means to within-unit variance, and fit a rough 2-type split at speed 1.0. If that looks clean, we can then do a small set of speed experiments to see whether speed scales rates multiplicatively.

import numpy as np
import pandas as pd

g = exp1.groupby("unit_id")["y"]
unit_means = g.mean().to_numpy()
unit_vars = g.var().to_numpy()
overall = exp1["y"].to_numpy()

print("overall mean:", overall.mean(), "overall var:", overall.var(ddof=1))
print("mean within-unit var:", np.mean(unit_vars))
print("var of unit means:", np.var(unit_means, ddof=1))

# simple 1D 2-means on unit means
x = unit_means.copy()
c1, c2 = np.quantile(x, [0.3, 0.7])
for _ in range(30):
    lab = np.abs(x - c1) <= np.abs(x - c2)
    if lab.all() or (~lab).all():
        break
    c1 = x[lab].mean()
    c2 = x[~lab].mean()
if c1 > c2:
    c1, c2 = c2, c1
    lab = ~lab

print("\n2-cluster centers on unit means:", c1, c2)
print("cluster sizes:", lab.sum(), (~lab).sum())

tmp = pd.DataFrame({"unit_mean": x, "cluster": np.where(lab, 0, 1)})
print("\ncluster summaries:")
print(tmp.groupby("cluster")["unit_mean"].describe())

# how well do 2 clusters explain unit mean variation?
pred = np.where(lab, c1, c2)
sse2 = np.sum((x - pred)**2)
sse1 = np.sum((x - x.mean())**2)
print("\nR^2 of 2-cluster approximation on unit means:", 1 - sse2/sse1)

# histogram-ish counts of unit means rounded to .5
rounded = pd.Series(np.round(x * 2) / 2).value_counts().sort_index()
print("\nrounded unit-mean frequencies:")
print(rounded.head(30))
