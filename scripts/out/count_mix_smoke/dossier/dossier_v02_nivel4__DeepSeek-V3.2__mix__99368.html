# Quick EM for two-component Poisson mixture (simplified)
y = df_hist['y'].values
n = len(y)
# Initialize
np.random.seed(42)
pi = 0.5  # mixing proportion
lambda1 = np.mean(y[y < np.median(y)])
lambda2 = np.mean(y[y >= np.median(y)])
print(f"Initial lambda1={lambda1:.2f}, lambda2={lambda2:.2f}")
# EM iterations (5 iterations for simplicity)
for iter in range(5):
    # E-step: compute responsibilities
    p1 = pi * np.exp(-lambda1) * (lambda1 ** y) / np.math.factorial(y)  # but factorial may overflow, use Poisson PMF from scipy? Let's approximate with log
    # Use scipy if available, but we can approximate via logs
    from math import log, exp, lgamma
    def poisson_logpmf(k, lam):
        return k * log(lam) - lam - lgamma(k + 1)
    log_p1 = poisson_logpmf(y, lambda1)
    log_p2 = poisson_logpmf(y, lambda2)
    # log responsibility
    log_resp1 = log(pi) + log_p1
    log_resp2 = log(1-pi) + log_p2
    # normalize
    max_log = np.maximum(log_resp1, log_resp2)
