import pandas as pd

# Fit mean vs speed using all available data
all_rows = pd.concat([
    pd.DataFrame({"speed": 1.0, "y": df["y"].astype(float)}),
    pd.DataFrame({"speed": 0.8, "y": exp_lo["y"].astype(float)}),
    pd.DataFrame({"speed": 1.0, "y": exp_mid["y"].astype(float)}),
    pd.DataFrame({"speed": 1.2, "y": exp_hi["y"].astype(float)}),
], ignore_index=True)

agg = all_rows.groupby("speed")["y"].agg(["mean", "var", "count"])
print("aggregated by speed:")
print(agg)

# linear fit mean = a + b*speed
x = all_rows["speed"].to_numpy()
y = all_rows["y"].to_numpy()
b = np.cov(x, y, ddof=0)[0, 1] / np.var(x)
a = y.mean() - b * x.mean()

# proportional-through-origin fit mean = c*speed
c = (x @ y) / (x @ x)

print("\nlinear fit a,b:", a, b)
print("origin fit c:", c)
print("pred means linear:", {s: a + b*s for s in sorted(all_rows['speed'].unique())})
print("pred means origin:", {s: c*s for s in sorted(all_rows['speed'].unique())})
