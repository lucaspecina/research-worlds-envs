import numpy as np

    mu = mu_base * speed
    
    # Number of full lots (each with R measurements)
    n_full_lots = n // R
    remainder = n % R
    
    # Generate unit_ids: each lot gets a sequential integer, repeated R times

    y_vals = []

        # Draw lambda for this lot from Gamma(shape=r, scale=mu/r)
