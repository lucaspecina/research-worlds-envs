import numpy as np
import pandas as pd

def model(regime, n, seed):
    np.random.seed(seed)
    # Parameters estimated from data
    base_rate = 6.453571428571428   # mean defects at speed=1.0
    k = 133.79337924288913          # dispersion parameter (Gamma shape)
    
    speed = regime.config.get("speed", 1.0)
    repeats = regime.config.get("repeats_per_unit", 1)
