import numpy as np
import pandas as pd

# Define model based on current assumptions
def model(regime, n, seed):
    # Set random seed for reproducibility
    rng = np.random.default_rng(seed)
    
    speed = regime.config.get("speed", 1.0)
    repeats = regime.config.get("repeats_per_unit", 1)
    
    # Mean defect rate at speed=1 from historical data
    base_mean = 6.39
    # Exponent from rough fit
    k = 0.57
    mean_lambda = base_mean * (speed ** k)
    
    # Dispersion parameter r for gamma (shape)
    # Chosen such that variance ~ mean + mean^2/r matches observed
    r = 36.0
    
    # Number of lots (unit_ids)
    n_lots = (n + repeats - 1) // repeats  # ceil division
    # Generate lambda for each lot from Gamma(shape=r, scale=mean_lambda/r)
    lambda_lots = rng.gamma(shape=r, scale=mean_lambda / r, size=n_lots)
    
    # Generate measurements
    unit_ids = []
    ys = []
    for i, lam in enumerate(lambda_lots):
        # Number of measurements for this lot: repeats, except possibly last lot
        m = repeats if (i < n_lots - 1) else (n - i * repeats)
        ys_lot = rng.poisson(lam=lam, size=m)
        unit_ids.extend([float(i)] * m)
        ys.extend(ys_lot.tolist())
    
    return pd.DataFrame({'unit_id': unit_ids, 'y': ys})
